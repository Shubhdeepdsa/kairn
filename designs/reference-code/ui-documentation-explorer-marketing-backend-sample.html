System Overview

ML
@m-lindqvist
·
commit commit d4e92a1
· Updated May 24, 2025

The Marketing Backend orchestrates multi-channel audience targeting, behavioral segmentation, and dynamic campaign payload dispatching. It is constructed as an event-driven cluster utilizing asynchronous message ingestion, deterministic idempotency filters, and distributed Redis key-space notifications to ensure sub-50ms message processing at continuous peak loads exceeding 40,000 requests per second.

verified_user Architectural Directive: Distributed Tracing & Tenant Isolation

Every ingress request must propagate the standard OpenTelemetry context headers (traceparent, x-tenant-partition-id). Requests lacking a verifiable cryptographically signed tenant token will terminate at the Envoy perimeter prior to internal queue publication.

src/services/event-dispatcher.ts 
import { Kafka, Producer, CompressionTypes } from 'kafkajs';
import { TraceContext, injectSpan } from '@kairn/telemetry';

// Partition-keyed campaign egress envelope
export interface CampaignDispatchEvent<T = Record<string, unknown>> {
  tenantId: string;
  campaignId: string;
  audienceSegmentId: string;
  payload: T;
  idempotencyKey: string;
  timestamp: number;
}

export class EventDispatcher {
  private producer: Producer;

  constructor(private readonly kafkaCluster: Kafka) {
    this.producer = this.kafkaCluster.producer({ allowAutoTopicCreation: false });
  }

  public async publishDispatch(
    event: CampaignDispatchEvent,
    ctx: TraceContext
  ): Promise<void> {
    const traceHeaders = injectSpan(ctx);

    await this.producer.send({
      topic: `marketing.events.${event.tenantId}`,
      compression: CompressionTypes.GZIP,
      messages: [{
        key: event.campaignId,
        value: JSON.stringify(event),
        headers: {
          ...traceHeaders,
          'x-idempotency-key': event.idempotencyKey,
        },
      }],
    });
  }
}
